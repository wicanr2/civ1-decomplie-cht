M4-lite: STR# 載入 + 14 文明 master + 14 領袖肖像 + 中文 catalog

# M4-lite 交付

  src/data/strings.{c,h}   STR# resource 解析器
                            - Mac Pascal-string array -> C blob + offsets
                            - 30 個 CIV_STR_* enum 對應 spec 05 §5.6
                              (Tax / Lux / Tech / Improvements / People /
                               Terrains / Cities / Leaders / Governments /
                               Caravan / Army / King / Civilopedia /
                               Story / Space / Disk / File ...)
  src/data/civs.{c,h}      14 文明 master table builder
                            - 從 STR# 140/143/144/148/149 各 16 entry
                              slot 1..14 抽取
                            - KING00..13 sprite 對應 slot 1..14
                            - hardcoded zh-TW catalog
                              (14 文明 x 5 欄位 = 70 條翻譯)
                            - civ_civs_zh(slot, "leader"|"civ_sing"|
                              "civ_plur"|"adj_sing"|"adj_plur")

  tests/test_strings.c     對 Civdata0.RSC 驗:
                            - STR# 130 = 72 techs (Alphabet etc.)
                            - STR# 132 = 28 units (Settlers etc.)
                            - STR# 137 Credits =
                                SID MEIER + BRUCE SHELLEY +
                                PAUL L. ROWAN + WILLIAM F. DENMAN, JR.
                            - STR# 140 Leaders =
                                slot 1=Caesar / 2=Hammurabi /
                                **4=Ramesses (不是 Cleopatra)**
                            - 14 文明 master table 與 zh-TW catalog
                              對齊驗證
                              (slot 1=凱撒/羅馬, 4=拉美西斯/埃及)

  tests/test_demo_snapshot.c 升 M4
                            - 14 文明 7x2 grid
                            - 每格載 CIVDATA2 #500..513 (KING00..13)
                              領袖肖像 + 中文文明名 + 英文領袖名
